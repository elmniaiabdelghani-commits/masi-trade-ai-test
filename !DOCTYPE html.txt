            const stopLoss = parseFloat(document.getElementById('stop-loss').value) || 0;
            const takeProfit = parseFloat(document.getElementById('take-profit').value) || 0;

            const riskAmount = capital * (riskPercent / 100);
            const riskPerShare = Math.abs(entryPrice - stopLoss);
            const sharesQty = riskPerShare > 0 ? Math.floor(riskAmount / riskPerShare) : 0;
            const capitalUsed = sharesQty * entryPrice;
            const potentialProfit = sharesQty * (takeProfit - entryPrice);
            const potentialLoss = sharesQty * (entryPrice - stopLoss);
            const riskRatio = potentialLoss > 0 ? (potentialProfit / potentialLoss).toFixed(1) : '0';

            document.getElementById('risk-amount').textContent = riskAmount.toLocaleString('fr-FR', {maximumFractionDigits: 0}) + ' MAD';
            document.getElementById('shares-qty').textContent = sharesQty;
            document.getElementById('capital-used').textContent = capitalUsed.toLocaleString('fr-FR', {maximumFractionDigits: 0}) + ' MAD';
            document.getElementById('risk-ratio').textContent = '1:' + riskRatio;
            document.getElementById('potential-profit').textContent = '+' + potentialProfit.toLocaleString('fr-FR', {maximumFractionDigits: 0}) + ' MAD';
            document.getElementById('potential-loss').textContent = '-' + potentialLoss.toLocaleString('fr-FR', {maximumFractionDigits: 0}) + ' MAD';
        }

        // Initialize
        document.addEventListener('DOMContentLoaded', () => {
            initTradingView();
            initMiniCharts();
            calculatePosition();
        });

        // Smooth scroll
        document.querySelectorAll('a[href^="#"]').forEach(anchor => {
            anchor.addEventListener('click', function (e) {
                e.preventDefault();
                const target = document.querySelector(this.getAttribute('href'));
                if (target) target.scrollIntoView({ behavior: 'smooth', block: 'start' });
            });
        });
